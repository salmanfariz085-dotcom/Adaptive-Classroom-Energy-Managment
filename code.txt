#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

// -------- PIN DEFINITIONS --------
#define PIR_PIN 14
#define LDR_PIN 34
#define LM35_PIN 32

#define RELAY_FAN 26
#define RELAY_LIGHT 25

#define FAN_SWITCH 33
#define LIGHT_SWITCH 27

// -------- OLED --------
#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, -1);

// -------- TIMING --------
unsigned long lastUpdate = 0;
const int interval = 200;

// -------- MOTION FILTER --------
bool motionState = false;
unsigned long lastMotionTime = 0;
unsigned long lastHighTime = 0;

const unsigned long motionHoldTime = 5000;
const unsigned long lowConfirmTime = 500;

// -------- MODES --------
int fanMode = 0;   // 0=AUTO, 1=ON, 2=OFF
int lightMode = 0; // 0=AUTO, 1=ON, 2=OFF

bool lastFanSwitch = HIGH;
bool lastLightSwitch = HIGH;

// -------- SETUP --------
void setup() {

  Serial.begin(115200);

  pinMode(PIR_PIN, INPUT);
  pinMode(LDR_PIN, INPUT);
  pinMode(LM35_PIN, INPUT);

  pinMode(RELAY_FAN, OUTPUT);
  pinMode(RELAY_LIGHT, OUTPUT);

  pinMode(FAN_SWITCH, INPUT_PULLUP);
  pinMode(LIGHT_SWITCH, INPUT_PULLUP);

  digitalWrite(RELAY_FAN, HIGH);
  digitalWrite(RELAY_LIGHT, HIGH);

  delay(3000);

  Wire.begin(21, 22);

  if (!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    while (true);
  }

  display.clearDisplay();
  display.setTextSize(1);
  display.setTextColor(WHITE);
}

// -------- LOOP --------
void loop() {

  if (millis() - lastUpdate >= interval) {
    lastUpdate = millis();

    // -------- SWITCH HANDLING --------
    bool fanSw = digitalRead(FAN_SWITCH);
    bool lightSw = digitalRead(LIGHT_SWITCH);

    if (lastFanSwitch == HIGH && fanSw == LOW) {
      fanMode++;
      if (fanMode > 2) fanMode = 0;
      delay(200);
    }

    if (lastLightSwitch == HIGH && lightSw == LOW) {
      lightMode++;
      if (lightMode > 2) lightMode = 0;
      delay(200);
    }

    lastFanSwitch = fanSw;
    lastLightSwitch = lightSw;

    int rawMotion = digitalRead(PIR_PIN);

    // -------- MOTION FILTER --------
    if (rawMotion == HIGH) {
      lastMotionTime = millis();
      lastHighTime = millis();
    }

    if (rawMotion == LOW) {
      if (millis() - lastHighTime > lowConfirmTime) {
        if (millis() - lastMotionTime > motionHoldTime) {
          motionState = false;
        }
      }
    }

    if (millis() - lastMotionTime <= motionHoldTime) {
      motionState = true;
    }

    int lightValue = analogRead(LDR_PIN);

    int adcValue = analogRead(LM35_PIN);
    float voltage = adcValue * (3.3 / 4095.0);
    float temp = voltage * 100.0;

    // -------- FAN CONTROL --------
    if (fanMode == 1) {
      digitalWrite(RELAY_FAN, LOW); // ON
    }
    else if (fanMode == 2) {
      digitalWrite(RELAY_FAN, HIGH); // OFF
    }
    else {
      if (motionState && temp > 24) {
        digitalWrite(RELAY_FAN, LOW);
      } else {
        digitalWrite(RELAY_FAN, HIGH);
      }
    }

    // -------- LIGHT CONTROL --------
    if (lightMode == 1) {
      digitalWrite(RELAY_LIGHT, LOW); // ON
    }
    else if (lightMode == 2) {
      digitalWrite(RELAY_LIGHT, HIGH); // OFF
    }
    else {
      if (motionState && lightValue >= 1600) {
        digitalWrite(RELAY_LIGHT, LOW);
      } else {
        digitalWrite(RELAY_LIGHT, HIGH);
      }
    }

    // -------- OLED --------
    display.clearDisplay();

    display.setCursor(0, 0);
    display.print("Temp: ");   
    display.print(temp);
    display.println(" C");

    display.print("Motion: ");
    display.println(motionState ? "YES" : "NO");

    display.print("Light: ");
    display.println(lightValue);

    display.println("");

    // FAN STATUS
    if (fanMode == 1) display.println("FAN: MANUAL ON");
    else if (fanMode == 2) display.println("FAN: MANUAL OFF");
    else display.println((motionState && temp > 24) ? "FAN: AUTO ON" : "FAN: AUTO OFF");

    // LIGHT STATUS
    if (lightMode == 1) display.println("LIGHT: MANUAL ON");
    else if (lightMode == 2) display.println("LIGHT: MANUAL OFF");
    else display.println((motionState && lightValue >= 1600) ? "LIGHT: AUTO ON" : "LIGHT: AUTO OFF");

    display.display();

    // -------- SERIAL --------
    Serial.print("FanMode: "); Serial.print(fanMode);
    Serial.print(" | LightMode: "); Serial.print(lightMode);
    Serial.print(" | Temp: "); Serial.print(temp);
    Serial.print(" | Motion: "); Serial.println(motionState);
  }
}
